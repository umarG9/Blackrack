contract BRBNFTMarketplace is Ownable {
    struct Listing { uint tokenId; address seller; uint price; bool active; }
    mapping(uint => Listing) public listings;
    mapping(uint => address) public highestBidder;
    mapping(uint => uint) public highestBid;
    uint public floorPrice = 100 * 10**18; // 100 BRB floor
    uint public totalVolume;
    uint public marketplaceFee = 3; // 3% fee - Like OpenSea 2.5% + BRB 0.5%
    IERC721 public brbNFT;
    IERC20 public brbToken;
    event Listed(uint tokenId, address seller, uint price);
    event Sold(uint tokenId, address buyer, address seller, uint price, uint fee);
    event AuctionBid(uint tokenId, address bidder, uint bid);
    constructor(address _brbNFT, address _brbToken) Ownable(msg.sender) {
        brbNFT = IERC721(_brbNFT);
        brbToken = IERC20(_brbToken);
    }
    function listNFT(uint tokenId, uint price) external {
        require(brbNFT.ownerOf(tokenId) == msg.sender, "Not owner");
        require(price >= floorPrice, "Price < floor");
        listings[tokenId] = Listing(tokenId, msg.sender, price, true);
        brbNFT.transferFrom(msg.sender, address(this), tokenId);
        emit Listed(tokenId, msg.sender, price);
    }
    function buyNFT(uint tokenId) external {
        Listing memory listing = listings[tokenId];
        require(listing.active, "Not listed");
        uint fee = listing.price * marketplaceFee / 100;
        uint sellerAmount = listing.price - fee;
        brbToken.transferFrom(msg.sender, listing.seller, sellerAmount);
        brbToken.transferFrom(msg.sender, owner(), fee); // Fee to marketplace owner
        brbNFT.transferFrom(address(this), msg.sender, tokenId);
        totalVolume += listing.price;
        listings[tokenId].active = false;
        emit Sold(tokenId, msg.sender, listing.seller, listing.price, fee);
    }
    function bidAuction(uint tokenId) external payable {
        // Auction - Highest bid wins
        require(listings[tokenId].active, "Not auction");
        require(msg.value > highestBid[tokenId], "Bid too low");
        highestBidder[tokenId] = msg.sender;
        highestBid[tokenId] = msg.value;
        emit AuctionBid(tokenId, msg.sender, msg.value);
    }
    function getFloorPrice() external view returns (uint) {
        return floorPrice;
    }
    function getTotalVolume() external view returns (uint) {
        return totalVolume;
    }
}
